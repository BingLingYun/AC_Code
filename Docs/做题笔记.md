##### CF2006C

https://codeforces.com/contest/2006/problem/C

**思考过程**：抱着从特殊到一般，手玩两个数，发现差值为2的幂次时，最终才能生成合法序列，然后就从这方面出发，考虑一个区间内具有什么性质才会是合法序列，可能最终一个合法区间就是有两个差值为2的幂次，然后由这些数继续去覆盖其余数，我把它当作了类似于区间合并那种，但是怎么具体该怎么快速的计算整个区间是否合法，不可做，**坠落**

**正解**：将一个区间里面的数排序后，令 $b_i$ =$a_{i+1}$-$a_i$，差值中只要有偶数，就可以进行操作，所以最终的序列中的$b_i$都为奇数，同时，如果两个相邻的奇数如果不相同，便又可以进行操作，因为$\frac{a_{i+1}+a_{i-1}}{2} \neq a_i$，所以最终的扩展完的序列变成了，一个公差为奇数的等差数列。公差*d*是原序列所有差值的因子，所以是*gcd*，那么就变成求区间的gcd为$2^k$或者0的个数

**实现**：使用st表求出区间gcd，然后固定左端点，二分l表示多少个区间gcd为0的，二分r表示多少个区间gcd为$2^k$,一个为前缀，一个为后缀，向右的gcd很显然越来越小

```c++
#include<bits/stdc++.h>

#define i64 long long
#define pii pair<int,int>
#define pll pair<i64,i64>
#define int128 __int128_t
#define ull unsigned long long
using namespace std;

constexpr i64 mod=1e9+7;

struct ST{
    vector<vector<int>>f;
    int n;
    ST(){}
    ST(int n)
    {
        n-=1;
        this->n=n;
        init(n);
    }
    void init(int n)
    {
        f.resize(n+1,vector<int>(31,0));
    }
    int quiry(int l,int r)
    {
        if(l==r) return 0;
        r-=1;
        int ln=(int)(log(r-l+1)/log(2));
        return (__gcd(f[l][ln],f[r-(1<<ln)+1][ln]));
    }
    void build(vector<int>a)
    {
        for(int i=1;i<=n;i++)
        {
            f[i][0]=abs(a[i]-a[i+1]);
        }
        int len=(int)(log(n)/log(2));
        for(int j=1;j<=len;j++)
        {
            for(int i=1;i<=n-(1<<j)+1;i++)
            {
                f[i][j]=__gcd(f[i][j-1],f[i+(1<<(j-1))][j-1]);
            }
        }
    }
};
void solve()
{
    int n;cin>>n;
    vector<int>a(n+1);
    for(int i=1;i<=n;i++) cin>>a[i];
    ST rmq(n);rmq.build(a);
    i64 ans=0;
    set<int>st;
    for(int i=0;i<=30;i++) st.insert((1<<i));
    auto check = [&] (int pos,int op)
    {
        int l=pos,r=n,res=-1;
        while(l<=r)
        {
            int mid=l+r>>1;
            if(op==0)
            {
                if(rmq.quiry(pos,mid)!=0) r=mid-1;
                else res=mid,l=mid+1;
            }
            else
            {
                int now=rmq.quiry(pos,mid);
                if(st.find(now)!=st.end()) res=mid,r=mid-1;
                else l=mid+1; 
            }
        }
        return  res;
    };
    for(int i=1;i<=n;i++)
    {
        int l=check(i,0);
        int r=check(i,1);
        ans+=(l-i+1);
        if(r!=-1) ans+=n-r+1;
    }
    cout<<ans<<"\n";
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout<<fixed<<setprecision(15);

    int t;cin>>t;
    while(t--)
    {
        solve();
    }
    return 0;
}
```



##### CF1998E1

https://codeforces.com/contest/1998/problem/E1

**思考过程**：求每一个值是否可以通过向左吃或者向右吃，把所有的值消掉。这种问题最开始考虑就是二分求一下左侧的所有值，如果想大于等于那个值，需要至少从哪里开始的和是可以达成条件的，利用二分和前缀和实现，右侧也是同理。当我理所当然地认为只有两侧的值都不会超过当前位置时才可以达成条件，发现如果一个值先吃右侧没超过的，达成左侧超过的位置，也可以实现结果。所以什么情况是无解的呢，就是左侧的端点需要至少右侧R的位置，在这其中，如果有一个右端点，它的左侧L小于等于左侧端点的话，这两个左右端点里面的值是无解的，所以就转换成了线段问题。**easy，起飞**

**实现**：二分+前缀和实现向右和向左的最少个数，然后从左往右往set集合里面放置左端点，在每一个位置的时候，利用右端点去找到最小的set集合里面的左端点，利用差分进行区间赋值。

```c++
#include<bits/stdc++.h>

#define i64 long long
#define pii pair<int,int>
#define pll pair<i64,i64>
#define int128 __int128_t
#define ull unsigned long long
using namespace std;

constexpr i64 mod=1e9+7;

void solve()
{
    int n,x;cin>>n>>x;
    vector<int>a(n+1);
    vector<i64>sum(n+2);
    for(int i=1;i<=n;i++) cin>>a[i],sum[i]=sum[i-1]+a[i];
    vector<int>pre(n+2),suf(n+2);
    auto check = [&] (int pos,int op)
    {
        int l,r,res;
        if(op==0) l=pos+1,r=n,res=n+1;
        else l=1,r=pos-1,res=0;
    
        while(l<=r)
        {
            int mid=l+r>>1;
            if(op==0)
            {
                if(sum[mid]-sum[pos] >= a[pos]) res=mid,r=mid-1;
                else l=mid+1;
            }
            else
            {
                if(sum[pos-1]-sum[mid-1] >= a[pos]) res=mid,l=mid+1;
                else r=mid-1;
            }
        }
        return res;
    };
    for(int i=1;i<=n;i++)
    {
        pre[i]=check(i,0);
        suf[i]=check(i,1);
    }
    vector<int>vis(n+1),d(n+2);
    set<int>st;
    vector<vector<int>>vec(n+1);
    for(int i=1;i<=n;i++)
    {
        if(pre[i]==n+1) d[i+1]+=1;
        else
        {
            vec[pre[i]].push_back(i);
        }
        if(suf[i]!=0 && st.size())
        {
            auto it=st.lower_bound(suf[i]);
            if(it!=st.end())
            {
                int l=*it;
                d[l+1]+=1;
                d[i]-=1;
            }
        }
        st.insert(i);
        for(auto x:vec[i]) st.erase(x);
    }
    for(int i=1;i<=n;i++) d[i]+=d[i-1];
    for(int i=1;i<=n;i++) if(d[i]>0) vis[i]=1;
    for(int i=1;i<=n;i++)
    {
        if(pre[i]==n+1)
        {
            for(int j=i+1;j<=n;j++) vis[j]=1;
            break;
        }
    }
    for(int i=n;i>=1;i--)
    {
        if(suf[i]==0)
        {
            for(int j=i-1;j>=1;j--) vis[j]=1;
            break;
        }
    }
    int ans=0;
    for(int i=1;i<=n;i++) ans+=vis[i];
    cout<<n-ans<<"\n";
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout<<fixed<<setprecision(15);

    int t;cin>>t;
    while(t--)
    {
        solve();
    }
    return 0;
}
```

##### CF1995D

https://codeforces.com/contest/1995/problem/D

**思考过程**：每k个字符必须选择一个，所以考虑是否可以通过状压枚举每一个情况，然后从前往后去进行dp，发现复杂度爆炸，飘过很多种暴力做法，都是$ 2^{36} $时间复杂度。**不可做，坠落**

**正解**：不超过k个字符必须有一个需要选中，所以考虑每一个位置往后k个字符以内的，放进一个字符集里面，有n-k+1种字符集。考虑一个合法的解，就是与每一个字符集都有交集。正着求不好求，那就反过来求非法解。求与字符集没有交集的，假如字符集为T，那么非法解是*U-T*的子集，所以求出所有非法集以及他们的子集，标记一下，答案就是最小的没有标记的。利用*SOS DP* 进行求解子集。

**实现**：算出所有长度为k的字符串，然后标记非法状态，同时再标记一下最后一个字母，最后一个字母肯定是末尾字母。然后利用*SOS DP* 进行优化求解。

```c++
#include<bits/stdc++.h>

#define i64 long long
#define pii pair<int,int>
#define pll pair<i64,i64>
#define int128 __int128_t
#define ull unsigned long long
using namespace std;

constexpr i64 mod=1e9+7;

void solve()
{
    int n,c,k;cin>>n>>c>>k;
    string s;cin>>s;s=" "+s;
    vector<int>cnt(c),p(1<<c);
    int now=0;
    for(int i=1;i<=k;i++)
    {
        int op=s[i]-'A';
        cnt[op]++;now|=(1<<op);
    }
    p[(1<<c)-1 - now]=1;
    for(int i=k+1;i<=n;i++)
    {
        int op1=s[i-k]-'A',op2=s[i]-'A';
        cnt[op2]++;now|=(1<<op2);
        cnt[op1]--;
        if(!cnt[op1]) now-=(1<<op1);
        
        p[(1<<c)-1 - now]=1;
    }
    p[(1<<c)-1 - (1<<(s[n]-'A'))]=1;
    for(int i=0;i<c;i++)
    {
        for(int j=0;j<(1<<c);j++) if(j>>i&1)
        {
            p[j-(1<<i)]|=p[j];
        }
    }
    int ans=c;
    for(int i=0;i<(1<<c);i++) 
    {
        if(!p[i]) ans=min(ans,__builtin_popcount(i));
    }
    cout<<ans<<"\n";
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout<<fixed<<setprecision(15);

    int t;cin>>t;
    while(t--)
    {
        solve();
    }
    return 0;
}
```



##### CF1855D

https://codeforces.com/contest/1855/problem/D

**思考过程**：对于每一个位置，如果前面有值的组合至少能达到当前位置，就去求一个最小的价值，那么当前的答案就是前缀和减去最小价值，然后每次利用最小价值去更新后面的值，我是利用线段树去更新的区间最小值，但是前面的情况更像是一种背包问题。然后线段树每次取最小值然后继续更新最小值，假了。因为这样可能有些值更新不到，sad，背包没想到利用bit进行优化，没有深究。**不可做，坠落**

**正解**：显然如果刚好unlock到*x*的话，那么答案就是前缀和减去*x-1*，所以从前往后就是去组合刚好到x的位置，显然就是dp问题，*dp[i]=dp[i-1] | dp[i-a[j] ]*，可以考虑利用bitset进行优化，f[j] |= f[j]<<a[i]，就是背包问题。需要注意的是，在转移 *i* 时，记录f[i]的真假之后就要把 *f[i]* 置为假，因为对于后面的值转移来说，前面刚好拼成的值是无法unlock到当前位置，也就没法利用前面的值进行转移。

**实现**：利用bitset优化背包，f[j] |= f[j]<<a[i]，f[j]进行左移a[i]位

```c++
#include<bits/stdc++.h>

#define i64 long long
#define pii pair<int,int>
#define pll pair<i64,i64>
#define int128 __int128_t
#define ull unsigned long long
#define ld long double
using namespace std;

constexpr i64 mod=998244353;

const int maxn=2e5;
bitset<maxn+10>bt;
int main()
{
	ios::sync_with_stdio(false);
    cin.tie(0);
    cout<<fixed<<setprecision(15);
    
    int n;cin>>n;
    vector<int>a(n+n+1);
    vector<i64>pre(n+n+1);
    for(int i=1;i<=n;i++) cin>>a[i];
    for(int i=1;i<=n+n;i++) pre[i]=pre[i-1]+a[i];
    bt[1]=1;
    int mx=1;
    i64 ans=0;
    vector<int>vis(n+n+1);
    for(int i=1;i<=n;i++)
    {
        if(mx<i) break;
        bt |= (bt<<a[i]);
        vis[i]=bt[i];mx=min(n,mx+a[i]);
        if(bt[i]) bt[i]=0;
    }
    for(int i=n+1;i<=n+n;i++) vis[i]=bt[i];
    for(int i=1;i<=n+n;i++) 
    {
        if(vis[i]) ans=max(ans,pre[i]-i+1);
    }
    cout<<ans<<"\n";
    return 0;
}
```



##### CF1982E

https://codeforces.com/contest/1982/problem/E

**思考过程**：看到这种题，区间范围这么大，数位dp？不对，要求的是区间。那么有什么规律吗，打表找找规律。NO！看了眼Hint，分治。怎么分治呢，可不可以从上到下求缩小范围来获得答案，不过这样得话还得记录一下左右满足条件的个数，以便区间合并时方便统计答案。**无了，没对想法进行深究**

**正解**：*f(n,k)*表示对*0 - n-1*求解k时返回的值，[l,r,ans]左边剩余和右边剩余，以及答案。通过拆分，f($2^m$,k)和 $2^m$ to *n*的答案。可以发现后者等价于f(n-$2^m$,k-1)，因为第m位一定为1，而前者可以通过拆分成f($2^{m-1},k)$ 和 f($2^{m-1}$ , k-1)。所以可以进行拆分，每次进行拆成两部分，然后合并。

**实现**：记忆化搜索，记录mp[x,y]=tuple(l,r,ans)，每次进行合并的时候，就类似于线段树中线段合并一样，*l* 和 *r* ，以及ans

```c++
#include <bits/stdc++.h>

#define i64 long long
#define pii pair<int, int>
#define pll pair<i64, i64>
#define int128 __int128_t
#define ull unsigned long long
using namespace std;

constexpr i64 mod = 1e9 + 7;

map<pair<i64, int>, tuple<i64, i64, i64>> mp;
map<i64, int> vis;
void solve()
{
    i64 n, k;
    cin >> n >> k;
    function<tuple<i64, i64, i64>(i64, int)> dfs = [&](i64 x, int y)
    {
        if (y < 0)
        {
            tuple now = {0ll, 0ll, 0ll};
            return now;
        }
        if (x == 1)
        {
            tuple now = {1ll, 1ll, 1ll};
            return now;
        }
        if (mp.find({x, y}) != mp.end())
            return mp[{x, y}];
        i64 lx, rx;
        if (vis[x])
            lx = rx = x / 2;
        else
        {
            i64 val = 1;
            for (int i = 0; i <= 60; i++)
            {
                i64 now = 1ll << i;
                if (now < x)
                    val = now;
                else
                    break;
            }
            lx = val, rx = x - val;
        }
        auto [l1, r1, ans1] = dfs(lx, y);
        auto [l2, r2, ans2] = dfs(rx, y - 1);
        i64 res = 1ll * (r1 % mod) * (l2 % mod) % mod + ans1 + ans2;
        res %= mod;
        i64 l = 0, r = 0;
        if (l1 == lx)
            l = l1 + l2;
        else
            l = l1;
        if (r2 == rx)
            r = r2 + r1;
        else
            r = r2;
        tuple ans = make_tuple(l, r, res);
        return mp[{x, y}] = ans;
    };
    auto [l, r, ans] = dfs(n, k);
    cout << ans << "\n";
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout << fixed << setprecision(15);

    for (int i = 0; i <= 60; i++)
    {
        i64 now = 1ll << i;
        vis[now] = 1;
    }
    int t;
    cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```

##### CF1966E

https://codeforces.com/contest/1966/problem/E

**思考过程**：想了一下折叠过程，两边的相同就是可以折叠，中间的需要三个可以折叠，相邻的01连续最终会被折叠掉（不知道为啥突然忽略条件跑到这一步了），最终生成0101相隔的字符串，尽可能多次折叠。最终好像理解错意思了（认为每一次折叠必须上下相同，分多步完成），可以一次性折叠，只要最终折叠在一起的字符相同即可，从上往下看那种。

**正解**：显然，折叠次数越多，答案越优，所以尽可以地多次折叠，那么每次遇见两个相邻且相同的字符进行折叠。折叠的最终结果一定是合法的，因为每次都翻转后面纸带上01的奇偶性，最终1和0会分别落在奇偶位置上，状态合法。所以模拟折叠过程即可。

**实现**：用l和r表示相对的左右边界位置，pos表示当前相对位置，依次去遍历字符，遇到相邻字符就反转方向。第一次翻转，后面纸带翻到前面纸带，翻转方向，当前位置往前；在遇到相邻，翻转方向，当前位置往后遍历，每次记录左右边界最远。

```c++
void solve()
{
    int n;

    int l = n + 1, r = 0;
    int ans = 0, dir = 1;
    for (int i = 1; i <= n; i++)
    {
        if (s[i] == s[i - 1])
            dir = -dir;
        else
            ans += dir;
        l = min(l, ans);
        r = max(r, ans);
    }
    cout << (r - l + 1) << "\n";
}

    }
    return 0;
```

##### CF1917D

https://codeforces.com/contest/1917/problem/D

**思考过程**：最终序列是每一个值依次去乘以2的幂次，可以发现当一个值比另外一个值多乘以$2^{21}$及以上肯定会更大，所以设置最多$m=20$次，当一个值选择一个幂次乘的话，大于当前幂次20次以上的都可以形成逆序，所以形成了一个等差求和序列，$(k-m-1)+...+1$是幂次相差大于m的个数和。然后考虑幂次差小于等于m的，依次去枚举幂次比现在小于$0,1,2...m$，那么就是当前数乘以这些幂次，大于这个值的话才能实现上面约束。同理依次去枚举幂次比现在大于$1,2..m$，那么就是当前数除以这些幂次，大于这个值才能实现，记得每个数本身之间的逆序对。

**实现**：BIT实现求逆序对，然后去实现上述过程

```c++
#include<bits/stdc++.h>

#define i64 long long
#define ull unsigned long long
#define pii pair<int,int>
#define pll pair<i64,i64>

using namespace std;
const i64 mod=998244353;

struct BIT{
    vector<int>a;int M;
    BIT (int N)
    {
        a=vector<int>(N+10,0);
    	M=N+5;
    }
    void add(int x,int v)
    {
        for(int i=x;i<=M;i+=(i&(-i))) a[i]+=v;
    }
    int ask(int x)
    {
        int res=0;
        for(int i=x;i;i-=(i&(-i))) res+=a[i];
        return res;
    }
    void resiz(int N)
    {
        a.clear();
        a.resize(N+10);
        M=N+5;
    }
    int quiry(int l,int r)
    {
        return ask(r)-ask(l-1);
    }
};
void solve()
{
    int n,k;cin>>n>>k;
    vector<int>a(n),b(k+1);
    for(int i=0;i<n;i++) cin>>a[i];
    for(int i=1;i<=k;i++) cin>>b[i],b[i]++;
    i64 ans=0;
    BIT fenwick(k);
    for(int i=1;i<=k;i++)
    {
        ans+=fenwick.quiry(b[i]+1,k);ans%=mod;
        fenwick.add(b[i],1);
    }
    ans=ans*n%mod;
    
    int m=min(20,k);
    fenwick.resiz(2*n+5);
    for(int i=0;i<n;i++)
    {
        i64 x=a[i];
        ans+=(i64)(1LL*(k-m-1)*(k-m)/2)%mod*i%mod;ans%=mod;
        for(int j=0;j<=m;j++)
        {
            i64 need=min((1LL<<j)*x,2LL*n);
            if(need%2==0) need+=1;
            if(need<=2*n) ans+=1LL*(k-j)*fenwick.quiry(need,2*n)%mod;
            ans%=mod;
        }
        for(int j=1;j<=m;j++)
        {   
            int need=x/(1<<j)+1;
            if(need%2==0) need+=1;
            if(need<=2*n) ans+=1LL*(k-j)*fenwick.quiry(need,2*n)%mod;
            ans%=mod;
        }
        fenwick.add(x,1);
    }
    cout<<ans<<"\n";
}

int main()
{
	ios::sync_with_stdio(false);
    cin.tie(0),cout.tie(0);
    cout<<fixed<<setprecision(20);
    
    int t;cin>>t;
    while(t--)
    {
        solve();
    }
    return 0;    
}
```

##### CF1825D2

https://codeforces.com/contest/1825/problem/D2

**思考过程**：看到这种，想着能不能从特殊到一般进行计算，k=1的时候。只有1个节点;k=2的时候，两个节点中间所有的节点都是，所以想着算出所有路径长，利用换根DP？很明显想歪了;k=3的时候，其实手摸一下，发现还是只有1个节点，但是没继续细想，力不足了。其次，没有往期望的性质上去思考，去想每个节点作为一个好节点的概率。

**正解**：k为奇数时，容易发现只有一个好节点，因为从一个节点移动到另一个节点时，没法平均分配，加一个奇数，减一个偶数，所以会有一个中心。k为偶数时，当一个子树里面选择$k/2$个节点时，在除子树外的节点上同样选择$k/2$个节点，连接节点u和它父亲的节点边是好边的概率$p[i]=\frac{\binom{S}{k/2}\binom{n-S}{k/2}}{\binom{n}{k}}$，最终期望好点的个数是期望好边的个数+1

**实现**：dfs遍历

```c++
#include <bits/stdc++.h>

#define i64 long long
#define pii pair<int, int>
#define pll pair<i64, i64>
#define int128 __int128_t
#define ull unsigned long long
using namespace std;

constexpr i64 mod = 1e9 + 7;

i64 fpow(i64 x, i64 y)
{
    i64 ans = 1;
    for (; y > 0; y >>= 1)
    {
        if (y & 1)
            ans = ans * x % mod;
        x = x * x % mod;
    }
    return ans;
}
struct Combi
{
    vector<i64> fac, inv;
    int n;
    Combi() {}
    Combi(int n)
    {
        init(n);
    }
    void init(int n)
    {
        this->n = n;
        fac.resize(n + 1, 0);
        inv.resize(n + 1, 0);
    }
    void init_C()
    {
        fac[0] = 1;
        for (int i = 1; i <= n; i++)
            fac[i] = fac[i - 1] * i % mod;
        inv[n] = fpow(fac[n], mod - 2);
        for (int i = n - 1; i >= 0; i--)
            inv[i] = inv[i + 1] * (i + 1) % mod;
    }
    i64 C(i64 x, i64 y)
    {
        if (x < y)
            return 0ll;
        return (fac[x] * inv[y] % mod * inv[x - y] % mod);
    }
};
void solve()
{
    int n, k;
    cin >> n >> k;
    Combi ca(n);
    ca.init_C();
    vector<vector<int>> g(n + 1);
    for (int i = 1; i < n; i++)
    {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    if (k % 2)
    {
        cout << 1 << "\n";
        return;
    }
    vector<i64> p(n + 1);
    vector<int> siz(n + 1);
    function<void(int, int)> dfs = [&](int u, int fa)
    {
        siz[u] = 1;
        for (auto v : g[u])
            if (v != fa)
            {
                dfs(v, u);
                siz[u] += siz[v];
            }
        p[u] = ca.C(siz[u], k / 2) * ca.C(n - siz[u], k / 2) % mod;
        p[u] = p[u] * fpow(ca.C(n, k), mod - 2) % mod;
    };
    dfs(1, 0);
    i64 ans = 1;
    for (int i = 1; i <= n; i++)
        (ans += p[i]) %= mod;
    cout << ans << "\n";
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout << fixed << setprecision(15);

    int t;
    t = 1;
    // cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```

##### CF1771F

https://codeforces.com/contest/1771/problem/F

**实现**：使用随机化为每个值赋值，进行异或求值，当一个区间有出现奇数次的数字是，区间异或不为零，每次对权值进行主席树上二分，主席树是权值线段树的实现

```c++
#include <bits/stdc++.h>

#define i64 long long
#define pii pair<int, int>
#define pll pair<i64, i64>
#define int128 __int128_t
#define ull unsigned long long
#define ld long double
using namespace std;

constexpr i64 mod = 998244353;
const int inf=INT32_MAX;

struct Segtree
{
    #define M(l,r) (l+(r-l)/2)
    vector<i64> sum, rt, ls, rs;
    int tot = 0;
    Segtree() {}
    Segtree(int n)
    {
        init(n);
    }
    void init(int n)
    {
        int maxn = (n << 5) + 10;
        sum.resize(maxn);
        rt.resize(maxn);
        ls.resize(maxn);
        rs.resize(maxn);
    }
    int update(int k, i64 val,int l, int r, int root) // 插入操作
    {
        int dir = ++tot;
        ls[dir] = ls[root], rs[dir] = rs[root], sum[dir] = sum[root] ^ val;
        if (l == r)
            return dir;
        int mid = M(l,r);
        if (k <= mid)
            ls[dir] = update(k, val, l, mid, ls[dir]);
        else
            rs[dir] = update(k, val, mid + 1, r, rs[dir]);
        return dir;
    }
    int query(int u, int v, int l, int r)// 查询操作
    { 
        int mid = M(l,r);
        i64 x = sum[ls[v]] ^ sum[ls[u]]; // 通过异或权值来获得区间是否存在奇数次数的数
        if (l == r)
            return l;
        if (x != 0)
            return query(ls[u], ls[v], l, mid);
        else
            return query(rs[u], rs[v], mid + 1, r);
    }
};

mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());//生成64位的随机数

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout << fixed << setprecision(15);

    int n;cin>>n;
    vector<int>a(n+1);
    for(int i=1;i<=n;i++) cin>>a[i];
    map<int,i64>mp;
    Segtree segtree(200000);

    for(int i=1;i<=n;i++)
    {
        if(mp.find(a[i])==mp.end()) mp[a[i]]=rng();
        segtree.rt[i]=segtree.update(a[i],mp[a[i]],0,inf,segtree.rt[i-1]);
    }
    int lst=0;
    int q;cin>>q;
    while(q--)
    {
        int x,y;cin>>x>>y;
        int l=x^lst,r=y^lst;
        int ans=segtree.query(segtree.rt[l-1],segtree.rt[r],0,inf);
        if(ans==inf) ans=0;
        cout<<ans<<"\n";
        lst=ans;
    }
    return 0;
}
```



##### ABC271G

https://atcoder.jp/contests/abc271/tasks/abc271_g

**正解**：令$f(i,j)$为第i次登录在第j个时间点，考虑怎么转移，发现$f_{i,j} = \sum_{k=1}^{24} f_{i-1,k}*dp_{k,j}$，从k节点转移过来，$dp_{i,j}$表示上一个访问时间点为i，下一个访问时间点为j的概率。$dp_{i,j} = \sum_{k=0}^{\infty} P^k*R*p_{j}$表示从$i+1$到$j-1$的都不选，然后其中可能会有n天概率一个都不访问。得到$f(i,j)$的求法之后，发现n的范围很大，但是知道转移方程，所以考虑矩阵加速。

$\begin{bmatrix}
f(n,1) & f(n,2) & \cdots & f(n,24)
\end{bmatrix}
=
\begin{bmatrix}
0 & 0 & \cdots & 1
\end{bmatrix}*\left(
\begin{bmatrix}
dp(1,1) & dp(1,2) & \cdots & dp(1,24) \\
dp(2,1) & dp(2,2) & \cdots & dp(2,24) \\
\vdots & \vdots & \ddots & \vdots \\
dp(24,1) & dp(24,2) & \cdots & dp(24,24)
\end{bmatrix}
\right)^n$

```c++
#include <bits/stdc++.h>

#define i64 long long
#define pii pair<int, int>
#define pll pair<i64, i64>
#define int128 __int128_t
#define ull unsigned long long
#define ld long double
using namespace std;

constexpr i64 mod = 998244353;

i64 fpow(i64 x,i64 y)
{
    i64 ans=1;
    for(;y>0;y>>=1)
    {
        if(y&1) ans=ans*x%mod;
        x=x*x%mod;
    }
    return ans;
}
//矩阵快速幂的式子一般是递推式，关键在于构建关系矩阵
i64 kpow(i64 x,i64 y)
{
    i64 res=0;
    for(;y>0;y>>=1)
    {
        if(y&1) res=(res+x)%mod;
        x=(x+x)%mod;
    }
    return res;
}//龟速乘，计算x*y

int L;//矩阵的大小，记得初始化
struct mul{
    i64 v[105][105];
    void clear(){ memset(v,0,sizeof(v)); }
    void reset()
    {
        clear();
        for(int i=1;i<=L;i++)
        {
            v[i][i]=1;//单位矩阵，相当与1
        }
    }
    mul friend operator * (const mul &a, const mul &b)//重载运算符
    {
        mul ret;ret.clear();
        for(int i=1;i<=L;i++)
        {
            for(int k=1;k<=L;k++)
            {
                for(int j=1;j<=L;j++)
                {
                    ret.v[i][j]+=kpow(a.v[i][k]%mod,b.v[k][j]%mod)%mod;
                    ret.v[i][j]%=mod;
                }
            }
        }
        return ret;
    }//矩阵乘法
};
inline mul fpow(mul base,i64 k)
{//k个base矩阵相乘
    mul ret;ret.reset();//ret为初始矩阵
    while(k)
    {
        if(k&1) ret=ret*base;
        base=base*base;
        k>>=1;
    }
    return ret;
}//快速幂
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout << fixed << setprecision(15);

    i64 n,x,y;cin>>n>>x>>y;
    string s;cin>>s;s=" "+s;
    int m=24;i64 P=1;
    i64 inv=fpow(100,mod-2);
    vector<i64>p(m+1);
    for(int i=1;i<=m;i++)
    {
        if(s[i]=='T') p[i]=x;
        else p[i]=y;
        P=P*(100-p[i])%mod*inv%mod;
    }
    vector dp(m+1,vector<i64>(m+1));
    for(int i=1;i<=m;i++)
    {
        for(int j=1;j<=m;j++)
        {
            i64 R=1;
            if(i<j)
            {
                for(int k=i+1;k<j;k++) 
                {
                    R=R*(100-p[k])%mod*inv%mod;
                }
            }
            else
            {
                for(int k=i+1;k<=m;k++)
                {
                    R=R*(100-p[k])%mod*inv%mod;
                }
                for(int k=1;k<j;k++)
                {
                    R=R*(100-p[k])%mod*inv%mod;
                }
            }
            dp[i][j]=1ll*R*(1ll*p[j]*inv%mod)%mod*fpow((1-P+mod)%mod,mod-2)%mod;
        }
    }
    L=24;
    mul base;
    for(int i=1;i<=L;i++)
    {
        for(int j=1;j<=L;j++)
        {
            base.v[i][j]=dp[i][j];
        }
    }
    mul A; A.v[1][L]=1;
    mul ans = A * fpow(base,n);

    i64 res=0;
    for(int i=1;i<=L;i++)
    {
        if(s[i]=='A') (res+=ans.v[1][i])%=mod;
    }
    cout<<res<<"\n";
    return 0;
}
```
